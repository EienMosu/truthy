- [ ] **Step 23: Write the stub and alias it in Vitest**

Create `tests/stubs/next-font-google.ts`:

```ts
// next/font/google only works inside the Next.js compiler. Vitest swaps it for this stub
// (see vitest.config.ts) so that components and the root layout can be rendered in tests.
interface StubFont {
  className: string;
  variable: string;
  style: { fontFamily: string };
}

function stubFont(name: string): (options?: unknown) => StubFont {
  return () => ({ className: name, variable: `${name}-variable`, style: { fontFamily: name } });
}

export const Overpass = stubFont("font-overpass");
export const Overpass_Mono = stubFont("font-overpass-mono");
```

  | xargs -0 perl -pi -e 's#\Q$ENV{HOME}\E/Desktop/(cloud-cards|truthy)/##g; s#/private[/]tmp/claude-[0-9]+/[^/\s"]+/[^/\s"]+/scratchpad/#<scratchpad>/#g; s#\Q$ENV{HOME}\E/#~/#g; s#~/\.claude/projects/-Users-[^/\s]+/[^/\s]+/tool-results/#<session tool-results>/#g'

p = pathlib.Path("content/save-pasted.py")

old = "TRANSCRIPTS = os.path.expanduser('~/.claude/projects/" + encoded_home + "/*.jsonl')\n"
